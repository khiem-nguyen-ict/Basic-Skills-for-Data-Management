In [246]:
# slices in steps
print(np1[1:6])
# start 1 Begin at index 1 (the 2nd element, zero-based).
# stop 6 Stop before index 6 (exclusive).
# step 2 Advance by 2 positions each time — take every other element.
print(np1[1:6:2]) # some data in steps of 2
# start from 0, until end, step 2
print(np1[::2]) # all data in steps of 2

[2 3 4 5 6]
[2 4 6]
[1 3 5 7 9]
